60// GPIO 6-11 drive the internal flash; 16/17 drive the WROVER's PSRAM, which holds the camera frame buffers.
61constexpr bool is_reserved(uint8_t p) { return (p >= 6 && p <= 11) || p == 16 || p == 17; }
62
63constexpr bool any_reserved(unsigned i = 0)
64{
65 return i < TURRET_COUNT && (is_reserved(TURRET[i]) || any_reserved(i + 1));
66}
67
68constexpr bool on_camera(uint8_t p, unsigned i = 0)
69{
70 return i < CAMERA_COUNT && (CAMERA[i] == p || on_camera(p, i + 1));
71}
72
73constexpr bool any_on_camera(unsigned i = 0)
74{
75 return i < TURRET_COUNT && (on_camera(TURRET[i]) || any_on_camera(i + 1));
76}

78constexpr bool any_duplicate(unsigned i = 0, unsigned j = 1)
79{
80 return i < TURRET_COUNT &&
81 (j < TURRET_COUNT ? (TURRET[i] == TURRET[j] || any_duplicate(i, j + 1)) : any_duplicate(i + 1, i + 2));
82}
