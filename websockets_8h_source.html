73 static esp_err_t websocket_handler(httpd_req_t* req)
74 {
75 CommandsWebSocketHandler* _this = reinterpret_cast<CommandsWebSocketHandler*>(req->user_ctx);
76
77 if (req->method == HTTP_GET)
78 {
79 // Initial WebSocket upgrade handshake: refuse pages from other origins before anything else.
80 if (!handshake_origin_ok(req, TAG_WEBSOCKETS))
81 {
82 return ESP_FAIL; // the server closes the session
83 }
84 ESP_LOGI(TAG_WEBSOCKETS, "WebSocket: New connection established to %s", _this->_uri);
85 return ESP_OK;
86 }
87
88 // Get the frame header (type and length) without reading the payload.
89 httpd_ws_frame_t ws_pkt;
90 memset(&ws_pkt, 0, sizeof(httpd_ws_frame_t));
91 esp_err_t ret = httpd_ws_recv_frame(req, &ws_pkt, 0);
92 if (ret != ESP_OK)
93 {
94 ESP_LOGW(TAG_WEBSOCKETS, "WebSocket: Frame receive error: %s", esp_err_to_name(ret));
95 return ret;
96 }
97
98 // The server answers PING/CLOSE itself (handle_ws_control_frames is off); do not reject one that slips through.
99 if (ws_pkt.type == HTTPD_WS_TYPE_PING || ws_pkt.type == HTTPD_WS_TYPE_PONG ||
100 ws_pkt.type == HTTPD_WS_TYPE_CLOSE)
101 {
102 return ESP_OK;
103 }
104 if (ws_pkt.type != HTTPD_WS_TYPE_TEXT || ws_pkt.len > kMaxMessage)
105 {
106 ESP_LOGW(TAG_WEBSOCKETS, "WebSocket: rejecting frame type=%d len=%u", (int)ws_pkt.type,
107 (unsigned)ws_pkt.len);
108 return ESP_FAIL; // the server closes the session
109 }
110
111 // A zero-length text frame has no payload to read but still gets its one reply (parse yields Poll).
112 uint8_t buf[kMaxMessage];
113 if (ws_pkt.len > 0)
114 {
115 ws_pkt.payload = buf;
116 ret = httpd_ws_recv_frame(req, &ws_pkt, ws_pkt.len);
117 if (ret != ESP_OK)
118 {
119 return ret;
120 }
121 }
122
123 if (_this->_remote != nullptr)
124 {
125 _this->_remote->submit(remote::parse(reinterpret_cast<const char*>(buf), ws_pkt.len));
126 }
127
128 char json[kReplySize];
129 const size_t json_len = _this->build_reply(json, sizeof(json));
130
131 httpd_ws_frame_t response_pkt = {0};
132 response_pkt.type = HTTPD_WS_TYPE_TEXT;
133 response_pkt.payload = reinterpret_cast<uint8_t*>(json);
134 response_pkt.len = json_len;
135 return httpd_ws_send_frame(req, &response_pkt);
136 }

140 size_t build_reply(char* out, size_t size) const
141 {
142 const int cap = (int)size;
143 if (this->_detection_instance == nullptr)
144 {
145 // Fallback if no detection module
146 ESP_LOGW(TAG_WEBSOCKETS, "No detection module connected, sending fallback metrics");
147 return (size_t)snprintf(out, size,
148 "{\"detected\":0,\"x\":0,\"y\":0,\"width\":320,\"height\":240,\"pixels\":0}");
149 }
150
151 const TrackSnapshot t = this->_detection_instance->get_track_snapshot();
152 // Existing keys first (older HUDs keep working), then the tracker fields.
153 int len =
154 snprintf(out, size,
155 "{\"detected\":%d,\"x\":%d,\"y\":%d,\"width\":%d,\"height\":%d,\"pixels\":%d,"
156 "\"state\":\"%s\",\"id\":%u,\"bx\":%d,\"by\":%d,\"bw\":%d,\"bh\":%d,"
157 "\"vx\":%.1f,\"vy\":%.1f",
158 t.detected() ? 1 : 0, t.x, t.y, t.frame_width, t.frame_height, t.area_px,
159 tracking::to_string(t.state), (unsigned)t.id, t.bx, t.by, t.bw, t.bh, (double)t.vx, (double)t.vy);
160 if (len > 0 && len < cap && this->_status_provider != nullptr)
161 {
162 const ControlStatus c = this->_status_provider->control_status();
163 const int extra =
164 snprintf(out + len, size - len,
165 ",\"mode\":\"%s\",\"cal\":%u,\"cal_state\":\"%s\",\"calibrated\":%d,"
166 "\"pan\":%.1f,\"tilt\":%.1f,\"cal_error\":\"%s\"",
167 to_string(c.mode), (unsigned)c.cal_points, cal_state_name(c.cal_state), c.calibrated ? 1 : 0,
168 (double)c.angles.pan, (double)c.angles.tilt, c.cal_failure ? c.cal_failure : "");
169 len = (extra < 0) ? -1 : len + extra;
170 }
171 if (len > 0 && len < cap - 1)
172 {
173 out[len++] = '}';
174 out[len] = '\0';
175 } else
176 {
177 len = -1;
178 }
179 if (len < 0 || len >= cap)
180 {
181 len = snprintf(out, size, "{\"detected\":0}");
182 }
183 return (size_t)len;
184 }
